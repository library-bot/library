function doPost(e) { try { const ss = SpreadsheetApp.getActiveSpreadsheet(); const borrowSheet = ss.getSheetByName("BORROW"); const booksSheet = ss.getSheetByName("BOOKS"); const data = JSON.parse(e.postData.contents); const code = String(data.code).trim(); const borrowerId = String(data.borrowerId).trim(); // TODAY DATE const today = new Date(); // ADD 5 WORKING DAYS let submissionDate = new Date(today); let count = 0; while (count < 5) { submissionDate.setDate( submissionDate.getDate() + 1 ); const day = submissionDate.getDay(); // Skip Friday and Saturday if (day !== 5 && day !== 6) { count++; } } // SEARCH BOOKS SHEET const booksData = booksSheet.getDataRange().getValues(); let bookName = ""; let price = ""; for (let i = 1; i < booksData.length; i++) { // BOOKS SHEET // A = S/L // B = CODE // C = BOOK NAME // D = STATUS // E = PRICE const bookCode = String(booksData[i][1]).trim(); if (bookCode === code) { bookName = booksData[i][2]; price = booksData[i][4]; // UPDATE STATUS booksSheet .getRange(i + 1, 4) .setValue("Borrowed"); break; } } // IF BOOK NOT FOUND if (bookName === "") { return ContentService .createTextOutput("Book Not Found") .setMimeType( ContentService.MimeType.TEXT ); } // SAVE TO BORROW SHEET borrowSheet.appendRow([ code, borrowerId, Utilities.formatDate( today, Session.getScriptTimeZone(), "dd-MMM-yyyy" ), Utilities.formatDate( submissionDate, Session.getScriptTimeZone(), "dd-MMM-yyyy" ), bookName, price, "Borrowed" ]); return ContentService .createTextOutput("Success") .setMimeType( ContentService.MimeType.TEXT ); } catch(err) { return ContentService .createTextOutput(err.toString()) .setMimeType( ContentService.MimeType.TEXT ); } }
