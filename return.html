function doGet() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName("BORROW"); const data = sheet.getDataRange().getValues(); let result = []; for (let i = 1; i < data.length; i++) { result.push({ code: data[i][0], borrowerId: data[i][1], borrowDate: data[i][2], submissionDate: data[i][3], bookName: data[i][4], price: data[i][5], status: data[i][6] }); } return ContentService .createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); }
