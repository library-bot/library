function doPost(e) { try { const ss = SpreadsheetApp.getActiveSpreadsheet(); const borrowSheet = ss.getSheetByName("BORROW"); const booksSheet = ss.getSheetByName("BOOKS"); const data = JSON.parse(e.postData.contents); const row = Number(data.row); const status = String(data.status).trim(); // update status borrowSheet.getRange(row, 7).setValue(status); const today = Utilities.formatDate( new Date(), Session.getScriptTimeZone(), "dd-MMM-yyyy" ); borrowSheet.getRange(row, 8).setValue(today); const values = borrowSheet.getRange(row, 1, 1, 8).getValues()[0]; const code = values[0]; const borrowerId = values[1]; const bookName = values[4]; const price = values[5]; const email = borrowerId + "@ncpsc.edu.bd"; let subject = ""; let headerColor = ""; let titleText = ""; // ========================= // STATUS THEMES // ========================= if (status === "Borrowed") { subject = "📘 Book Borrowed Confirmation"; headerColor = "linear-gradient(135deg,#2563eb,#1d4ed8)"; titleText = "📘 BORROWED CONFIRMATION"; } else if (status === "Returned") { subject = "📗 Book Return Confirmation"; headerColor = "linear-gradient(135deg,#16a34a,#22c55e)"; titleText = "📗 RETURN CONFIRMATION"; } else if (status === "Lost" || status === "Damaged") { subject = "⚠️ Book Issue Alert"; headerColor = "linear-gradient(135deg,#dc2626,#f97316)"; titleText = "⚠️ BOOK ISSUE ALERT"; } // ========================= // EMAIL BODY (COMMON DESIGN) // ========================= const mailBody = `

📚 NCPSC LIBRARY

${titleText}

Dear ${borrowerId},

Your library record has been updated successfully. Below are the details:

Book Name${bookName}
Code${code}
Status${status}
Date${today}

বাংলা একাডেমী প্রমিত বাংলা ভাষার ব্যাকরণ / ইসলাম, রফিকুল (সম্পাদক)

This is an auto-generated email. Do not reply.

`; // ========================= // SEND EMAIL // ========================= MailApp.sendEmail({ to: email, subject: subject, htmlBody: mailBody }); // ========================= // BOOK STATUS UPDATE // ========================= if (status === "Returned") { const booksData = booksSheet.getDataRange().getValues(); for (let i = 1; i < booksData.length; i++) { if (String(booksData[i][1]).trim() === String(code).trim()) { booksSheet.getRange(i + 1, 4).setValue("Available"); break; } } } if (status === "Borrowed") { const booksData = booksSheet.getDataRange().getValues(); for (let i = 1; i < booksData.length; i++) { if (String(booksData[i][1]).trim() === String(code).trim()) { booksSheet.getRange(i + 1, 4).setValue("Borrowed"); break; } } } return ContentService .createTextOutput("Success") .setMimeType(ContentService.MimeType.TEXT); } catch (err) { return ContentService .createTextOutput(err.toString()) .setMimeType(ContentService.MimeType.TEXT); } }
